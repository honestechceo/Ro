// RO Log reader + cloud backup (Cloudflare Worker)
// Settings > Variables and Secrets:  GOOGLE_API_KEY (secret), APP_TOKEN (secret), ALLOWED_ORIGINS (text)
// Settings > Bindings: KV namespace, variable name BACKUPS
export default {
  async fetch(req, env) {
    const origin = req.headers.get("origin") || "";
    const norm = (u) => { u = String(u || "").trim(); if (!u) return ""; if (!/^https?:\/\//i.test(u)) u = "https://" + u; try { return new URL(u).origin.toLowerCase(); } catch { return ""; } };
    const allowed = String(env.ALLOWED_ORIGINS || "").split(/[,\s]+/).map(norm).filter(Boolean);
    const originOk = !allowed.length || allowed.includes(norm(origin));
    const cors = { "Access-Control-Allow-Origin": origin || "*", "Access-Control-Allow-Methods": "POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type, X-App-Token", "Access-Control-Max-Age": "86400", Vary: "Origin" };
    const json = (o, s = 200) => new Response(JSON.stringify(o), { status: s, headers: { ...cors, "Content-Type": "application/json" } });
    if (req.method === "OPTIONS") return new Response(null, { status: 204, headers: cors });
    if (!originOk) return json({ error: `${origin || "This site"} isn't in ALLOWED_ORIGINS.` }, 403);
    const token = String(env.APP_TOKEN || "").trim();
    if (token && (req.headers.get("x-app-token") || "").trim() !== token) return json({ error: "Wrong passphrase." }, 401);
    if (req.method !== "POST") return json({ ok: true, message: "RO Log reader is running." });
    const key = env.GOOGLE_API_KEY;
    if (!key) return json({ error: "GOOGLE_API_KEY isn't set on Cloudflare." }, 500);
    let body;
    try { body = await req.json(); } catch { return json({ error: 'Send JSON: {"image": "<base64 jpeg>"}' }, 400); }
    const vision = (requests) => fetch(`https://vision.googleapis.com/v1/images:annotate?key=${encodeURIComponent(key)}`, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ requests }) });

    if (body && body.cloud) return cloud(body, token, env, json);
    if (body && body.ping) {
      const px = "iVBORw0KGgoAAAANSUhEUgAAAAEAAAABCAYAAAAfFcSJAAAADUlEQVR42mP8z8BQDwAEhQGAhKmMIQAAAABJRU5ErkJggg==";
      const t = await vision([{ image: { content: px }, features: [{ type: "TEXT_DETECTION", maxResults: 1 }] }]).catch(() => null);
      if (!t) return json({ error: "Couldn't reach Google Vision." }, 502);
      const d = await t.json().catch(() => ({}));
      if (!t.ok || d.error) return json({ error: "Google key problem: " + ((d.error && d.error.message) || `error ${t.status}`) }, 502);
      return json({ ok: true, cloud: !!env.BACKUPS });
    }
    const image = String((body && body.image) || "").replace(/^data:image\/\w+;base64,/, "");
    if (!image || image.length > 9_000_000) return json({ error: "Image missing or too large." }, 400);
    const g = await vision([{ image: { content: image }, features: [{ type: "DOCUMENT_TEXT_DETECTION" }], imageContext: { languageHints: ["en-t-i0-handwrit"] } }]);
    const data = await g.json().catch(() => ({}));
    if (!g.ok || data.error) return json({ error: (data.error && data.error.message) || `Google Vision error ${g.status}` }, 502);
    const res = (data.responses && data.responses[0]) || {};
    if (res.error) return json({ error: res.error.message }, 502);
    const box = (b) => { const v = ((b && b.vertices) || []).map((p) => ({ x: p.x || 0, y: p.y || 0 })); if (v.length < 4) return null; const xs = v.map((p) => p.x), ys = v.map((p) => p.y), x = Math.min(...xs), y = Math.min(...ys); return { x, y, w: Math.max(...xs) - x, h: Math.max(...ys) - y }; };
    const words = [];
    for (const page of (res.fullTextAnnotation && res.fullTextAnnotation.pages) || [])
      for (const block of page.blocks || [])
        for (const para of block.paragraphs || [])
          for (const word of para.words || []) {
            const t = (word.symbols || []).map((s) => s.text).join(""), b = box(word.boundingBox);
            if (!t || !b) continue;
            const s = (word.symbols || []).map((sy) => { const sb = box(sy.boundingBox); return sb ? { t: sy.text, ...sb } : null; }).filter(Boolean);
            words.push({ t, ...b, c: word.confidence || 0, s });
          }
    return json({ words, text: (res.fullTextAnnotation && res.fullTextAnnotation.text) || "" });
  },
};

// Encrypted RO Log backups. The phone locks the data with your passphrase before sending; this only stores the locked copy.
async function cloud(body, token, env, json) {
  if (!token) return json({ error: "Set APP_TOKEN on Cloudflare to use cloud backup." }, 400);
  if (!env.BACKUPS) return json({ error: "Add the BACKUPS storage binding on Cloudflare to use cloud backup." }, 400);
  const hash = await crypto.subtle.digest("SHA-256", new TextEncoder().encode("rolog|" + token));
  const id = [...new Uint8Array(hash)].map((b) => b.toString(16).padStart(2, "0")).join("").slice(0, 32);
  if (body.cloud === "save") {
    const { blob, iv, salt } = body;
    if (typeof blob !== "string" || typeof iv !== "string" || typeof salt !== "string") return json({ error: "Bad backup." }, 400);
    if (blob.length > 20_000_000) return json({ error: "Backup is too large." }, 413);
    const prevRaw = await env.BACKUPS.get(id + "/latest");
    if (prevRaw) { try { const p = JSON.parse(prevRaw); const keep = await env.BACKUPS.get(id + "/prev-ts"); if (!keep || Date.now() - +keep > 6 * 3600e3) { await env.BACKUPS.put(id + "/prev", prevRaw); await env.BACKUPS.put(id + "/prev-ts", String(Date.now())); } } catch {} }
    const ts = Date.now();
    await env.BACKUPS.put(id + "/latest", JSON.stringify({ blob, iv, salt, ts, n: +body.n || 0 }));
    return json({ ok: true, ts });
  }
  if (body.cloud === "load") { const v = await env.BACKUPS.get(id + "/latest"); return json(v ? JSON.parse(v) : { none: true }); }
  return json({ error: "Unknown cloud request." }, 400);
}
